<script>
    function checkStep1(btn, choice) {
        const fb = document.getElementById('feedback1');
        if(choice === '오전') {
            fb.innerText = "정답이에요! 아침은 오전입니다. ✨";
            btn.classList.add('correct');
        } else {
            fb.innerText = "다시 한번 생각해볼까요? 🧐";
            btn.classList.add('wrong');
            setTimeout(() => btn.classList.remove('wrong'), 1000);
        }
    }

    function checkStep2() {
        const val = document.getElementById('quiz2').value;
        const fb = document.getElementById('feedback2');
        if(val == 15) {
            fb.innerText = "와! 15시 맞아요. 계산 천재! 🎈";
        } else {
            fb.innerText = "3에 12를 더하면 얼마일까요?";
        }
    }

    function checkStep3() {
        const val = document.getElementById('quiz3').value;
        const fb = document.getElementById('feedback3');
        if(val == 5) {
            fb.innerText = "정답! 17에서 12를 빼면 오후 5시가 돼요! 🚄";
        } else {
            fb.innerText = "17시... 조금만 더 생각해봐요!";
        }
    }
</script>
